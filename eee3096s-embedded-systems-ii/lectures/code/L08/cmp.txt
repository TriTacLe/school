#include <stdio.h>

int main ()
{
    int a,b, res;
    int avg;
    scanf("%d %d",&a, &b);
    a = 100;
    b = 200;
    avg = (a+b)/2;
    if (a>avg) res=1;
    if (a<avg) res=-1;
    if (a==avg) res=0;
    if (res==1) printf("GRT\n");
    if (res==0) printf("EQU\n");
    if (res==-1) printf("EQU\n");
    printf("%d\n",res);
    return 0;
}
