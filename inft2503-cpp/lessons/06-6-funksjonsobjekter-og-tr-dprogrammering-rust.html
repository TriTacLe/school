6-funksjonsobjekter-og-trådprogrammering

Funksjonsobjekter og trådprogrammering i Rust

Funksjonsobjekter

Vi demonstrerer bruk av funksjonsobjekter i et eksempel der vi implementerer skallet til et GUI bibliotek. Biblioteket inneholder kun klassene for et Vindu, Window, og komponenter (widgets) som kan plasseres i vinduet: Image og Button.

Eksempel med overbruk av polymorfi

Før i tiden ble funksjonalitet for eksempel til en knapp lagt til en egen klasse (se MyButton) som arvet fra en abstrakt Button klasse:

#include <iostream>
#include <memory>
#include <string>
#include <vector>

class Widget {
public:
  virtual void draw() = 0;

  virtual ~Widget() {}
};

class Image : public Widget {
public:
  Image(std::string file_) : file(std::move(file_)) {}

  std::string file;

  void draw() override {
    std::cout << "Draw image from file: " << file << std::endl;
  }
};

class Button : public Widget {
public:
  Button(std::string label_) : label(std::move(label_)) {}

  std::string label;

  void draw() override {
    std::cout << "Draw button labeled: " << label << std::endl;
    std::cout << "This happens when the button is clicked:" << std::endl;
    on_click();
  }

  virtual void on_click() = 0;
};

class MyButton : public Button {
public:
  MyButton() : Button("Click me") {}

  void on_click() override {
    std::cout << "Button was clicked" << std::endl;
  }
};

class Window {
public:
  std::vector<std::unique_ptr<Widget>> widgets;

  void show() {
    for (auto &widget : widgets) {
      widget->draw();
    }
  }
};

int main() {
  Window window;

  window.widgets.emplace_back(new Image("image.png"));

  window.widgets.emplace_back(new MyButton());

  window.show();
}

Legg spesielt merke til for eksempel Image-konstruktøren: Image(std::string file_) : file(std::move(file_)) {}. Når en skal lagre et objekt i en klasse fra et konstruktør argument er det mest effektivt å flytte dataen til datamedlemmet med std::move. Ofte kan en unngå dette ved å ikke ha med konstruktører, og i stedet for opprette en ny instanse slik:

#include <string>

class Image {
public:
  std::string file;
};

int main() {
  auto image1 = Image{"image.png"};
  // or
  Image image2{"image.png"};
}

, men dette går ikke når for eksempel Image arver en annen klasse (Widget).

Det første kodeeksempelet ville vært svært vanskelig å implementere i Rust.

Eksempelet i C++ med funksjonsobjekt

Med funksjonsobjekter kan vi slippe å lage en ny klasse for å implementere en knapp:

#include <functional>
#include <iostream>
#include <memory>
#include <string>
#include <vector>

class Widget {
public:
  virtual void draw() = 0;

  virtual ~Widget() {}
};

class Image : public Widget {
public:
  Image(std::string file_) : file(std::move(file_)) {}

  std::string file;

  void draw() override {
    std::cout << "Draw image from file: " << file << std::endl;
  }
};

class Button : public Widget {
public:
  Button(std::string label_, std::function<void()> on_click_) : label(std::move(label_)), on_click(std::move(on_click_)) {}

  std::string label;
  std::function<void()> on_click;

  void draw() override {
    std::cout << "Draw button labeled: " << label << std::endl;
    if (on_click) {
      std::cout << "This happens when the button is clicked:" << std::endl;
      on_click();
    }
  }
};

class Window {
public:
  std::vector<std::unique_ptr<Widget>> widgets;

  void show() {
    for (auto &widget : widgets) {
      widget->draw();
    }
  }
};

int main() {
  Window window;

  window.widgets.emplace_back(new Image("image.png"));

  window.widgets.emplace_back(new Button(
      "Click me",
      [] {
        std::cout << "Button was clicked" << std::endl;
      }));

  window.show();
}

Her legger vi funksjonaliteten til knappen når den legges til i vinduet i main()-funksjonen.

Eksempelet i Rust med funksjonsobjekt

Forrige eksempel med bruk av funksjonsobjekter i C++ kan programmeres så og si direkte i Rust:

trait Drawable {
    fn draw(self: &Self) -> ();
}

struct Image {
    file: String,
}

impl Drawable for Image {
    fn draw(&self) {
        println!("Draw image from file: {}", self.file);
    }
}

struct Button {
    label: String,
    on_click: Box<dyn Fn() -> ()>,
}

impl Drawable for Button {
    fn draw(&self) {
        println!("Draw button labeled: {}", self.label);
        println!("This happens when the button is clicked:");
        (self.on_click)();
    }
}

struct Window {
    widgets: Vec<Box<dyn Drawable>>,
}

impl Window {
    fn show(self: &Self) {
        for widget in &self.widgets {
            widget.draw();
        }
    }
}

fn main() {
    let mut window = Window {
        widgets: Vec::new(),
    };

    window.widgets.push(Box::new(Image {
        file: String::from("image.png"),
    }));

    window.widgets.push(Box::new(Button {
        label: String::from("Click me"),
        on_click: Box::new(|| {
            println!("Button was clicked");
        }),
    }));

    window.show();
}

I Rust kan funksjonsobjekter bli lagret i typen Box<dyn Fn() -> ()>, og anonyme funksjoner har skrivemåten || {}. Parametre kan gis mellom | symbolene.

Trådprogrammering

Vi skal demonstrere trådprogrammering i Rust med mutex, og skal gjøre om følgende C++ kode:

#include <chrono>
#include <iostream>
#include <mutex>
#include <thread>
#include <unordered_map>

int main() {
  std::unordered_map<std::string, std::string> results;
  std::mutex results_mutex;

  std::thread t1([&results, &results_mutex, url = "https://www.ntnu.no/"] {
    // Simulate getting result from url
    std::this_thread::sleep_for(std::chrono::seconds(1));

    std::lock_guard<std::mutex> lock(results_mutex);
    results[url] = "Velkommen til NTNU";
  });

  std::thread t2([&results, &results_mutex, url = "https://www.mit.edu/"] {
    // Simulate getting result from url
    std::this_thread::sleep_for(std::chrono::seconds(1));

    std::lock_guard<std::mutex> lock(results_mutex);
    results[url] = "Welcome to MIT";
  });

  t1.join();
  t2.join();

  std::cout << results["https://www.ntnu.no/"] << std::endl;
  std::cout << results["https://www.mit.edu/"] << std::endl;
}

Eksempelet over kan ikke skrives om til Rust direkte, siden Rust bruker en utvidet Mutex klasse som i tillegg inneholder verdien en vil beskytte. En slik klasse kan vi også programmere i C++:

#include <chrono>
#include <iostream>
#include <mutex>
#include <thread>
#include <unordered_map>

/// Similar to Mutex class in Rust.
/// data and mutex combined, where the data cannot be access without locking the mutex.
template <typename T>
class Mutex {
  // T equals std::unordered_map<std::string, std::string> in the example in main()
  T data;
  std::mutex mutex;

public:
  class Locked {
    T &data;
    /// Store lock as pointer so that one can write: auto lock = results->lock();
    /// One cannot move or copy a std::lock_guard<std::mutex> object, but a pointer to this object can be moved/copied.
    std::unique_ptr<std::lock_guard<std::mutex>> lock;

  public:
    Locked(T &data_, std::mutex &mutex_) : data(data_), lock(new std::lock_guard<std::mutex>(mutex_)) {
    }

    /// Implements the *operator for data access
    T &operator*() {
      return data;
    }
  };

  /// Returns instance of Locked and locks the instance's mutex.
  Locked lock() {
    return Locked(data, mutex);
  }
};

int main() {
  // Mutex: data and mutex combined, where the data cannot be access without locking the mutex
  auto results = std::make_shared<Mutex<std::unordered_map<std::string, std::string>>>();

  std::thread t1([results /* Makes copy of results */, url = "https://www.ntnu.no/"] {
    // Simulate getting result from url
    std::this_thread::sleep_for(std::chrono::seconds(1));

    // Access the data by locking the Mutex object
    auto lock = results->lock();
    (*lock).emplace(url, "Velkommen til NTNU");

    // The Mutex object is unlocked at end of scope
  });

  std::thread t2([results /* Makes copy of results */, url = "https://www.mit.edu/"] {
    // Simulate getting result from url
    std::this_thread::sleep_for(std::chrono::seconds(1));

    // Access the data by locking the Mutex object
    auto lock = results->lock();
    (*lock).emplace(url, "Welcome to MIT");

    // The Mutex object is unlocked at end of scope
  });

  t1.join();
  t2.join();

  // Cannot access data without calling lock(), even though locking is unnecessary.
  auto lock = results->lock();
  std::cout << (*lock)["https://www.ntnu.no/"] << std::endl;
  std::cout << (*lock)["https://www.mit.edu/"] << std::endl;
}

Rust koden blir da svært likt forrige eksempel:

use std::collections::HashMap;
use std::sync::{Arc, Mutex};
use std::{thread, time};

fn main() {
    // Arc: tilsvarer shared_ptr i C++
    // Mutex: data and mutex combined, where the data cannot be access without locking the mutex
    let results = Arc::new(Mutex::new(HashMap::<String, String>::new()));

    let results_copy = results.clone();
    let url = "https://www.ntnu.no/".to_string();
    let t1 = thread::spawn(move || {
        // results_copy is moved here

        // Simulate getting result from url
        thread::sleep(time::Duration::from_secs(1));

        // Access the data by locking the Mutex object
        let mut lock = results_copy.lock().unwrap();
        (*lock).insert(url, "Velkommen til NTNU".to_string());

        // The Mutex object is unlocked at end of scope
    });

    let results_copy = results.clone();
    let url = "https://www.mit.edu/".to_string();
    let t2 = thread::spawn(move || {
        // results_copy is moved here

        // Simulate getting result from url
        thread::sleep(time::Duration::from_secs(1));

        // Access the data by locking the Mutex object
        let mut lock = results_copy.lock().unwrap();
        (*lock).insert(url, "Welcome to MIT".to_string());

        // The Mutex object is unlocked at end of scope
    });

    t1.join();
    t2.join();

    // Cannot access data without calling lock(), even though locking is unnecessary.
    let lock = results.lock().unwrap();
    println!("{}", (*lock)["https://www.ntnu.no/"]);
    println!("{}", (*lock)["https://www.mit.edu/"]);
}

I stedet for å spesifisere hva som skal fanges av den anonyme funksjonen i Rust, move || {}, blir objekter som blir brukt inne denne anonyme funksjonen automatisk fanget. Uttrykket move gjør at objektene som blir brukt inne i den anonyme funksjonen blir flyttet i stedet for å bli referert til.
